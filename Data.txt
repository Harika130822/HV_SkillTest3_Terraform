use("ecommerce");

const user = db.users.findOne({ isActive: { $ne: false } });
const products = db.products.find({ isActive: { $ne: false } }).limit(3).toArray();

if (!user) throw new Error("Create a user before inserting sample orders.");
if (!products.length) throw new Error("Insert products before inserting sample orders.");

const productAt = (index) => products[index % products.length];

const makeItem = (product, quantity) => ({
  productId: product._id,
  quantity,
  price: product.price,
  total: Number((product.price * quantity).toFixed(2)),
  productDetails: {
    name: product.name,
    sku: product.inventory.sku,
    image: product.images?.[0]?.url || ""
  }
});

const address = {
  firstName: "Alex",
  lastName: "Morgan",
  street: "123 Market Street",
  city: "San Francisco",
  state: "California",
  zipCode: "94105",
  country: "USA",
  phone: "555-0100"
};

const makeOrder = (number, items, status, paymentStatus, paymentMethod, shippingMethod) => {
  const subtotal = Number(items.reduce((sum, item) => sum + item.total, 0).toFixed(2));
  const tax = Number((subtotal * 0.08).toFixed(2));
  const shippingCost = 5.99;

  return {
    orderNumber: `ORD-SAMPLE-${Date.now()}-${number}`,
    userId: user._id,
    items,
    pricing: {
      subtotal,
      tax,
      shipping: shippingCost,
      discount: 0,
      total: Number((subtotal + tax + shippingCost).toFixed(2))
    },
    shippingAddress: address,
    billingAddress: address,
    status,
    paymentStatus,
    paymentMethod,
    paymentDetails: paymentStatus === "paid"
      ? {
          transactionId: `TXN-SAMPLE-${number}`,
          paymentGateway: paymentMethod,
          paidAt: new Date()
        }
      : {},
    shipping: {
      method: shippingMethod,
      carrier: status === "shipped" || status === "delivered" ? "Sample Express" : undefined,
      trackingNumber: status === "shipped" || status === "delivered"
        ? `TRACK-SAMPLE-${number}`
        : undefined,
      estimatedDelivery: new Date(Date.now() + 5 * 24 * 60 * 60 * 1000)
    },
    customerNotes: "Sample order for testing",
    createdAt: new Date(),
    updatedAt: new Date()
  };
};

const result = db.orders.insertMany([
  makeOrder(
    1,
    [makeItem(productAt(0), 1), makeItem(productAt(1), 2)],
    "shipped",
    "paid",
    "credit_card",
    "standard"
  ),
  makeOrder(
    2,
    [makeItem(productAt(1), 1)],
    "processing",
    "pending",
    "cash_on_delivery",
    "express"
  ),
  makeOrder(
    3,
    [makeItem(productAt(2), 2)],
    "delivered",
    "paid",
    "paypal",
    "standard"
  )
]);

printjson(result);
